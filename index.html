// Android Studio (Kotlin) Tarafı İçin Otomatik Gizlenen Üst Bar Mantığı webView.setOnScrollChangeListener { _, _, scrollY, _, oldScrollY -> if (scrollY > oldScrollY + 10) { // Aşağı kaydırıldı -> Üst barı gizle toolbar.animate().translationY(-toolbar.height.toFloat()).setDuration(200).start() } else if (scrollY < oldScrollY - 10) { // Yukarı kaydırıldı -> Üst barı göster toolbar.animate().translationY(0f).setDuration(200).start() } }
